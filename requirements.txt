# Runtime dependencies. Installed both locally and in the deployment image.
#
# Kept at the repository root because that is where every deployment platform
# looks first, including Render's native Python build. Test-only packages live
# in requirements-dev.txt so the production image does not carry a browser
# automation stack.
#
# Lower bounds rather than exact pins: this is an application with a small,
# stable dependency set, and the geometry tests would catch a breaking change in
# the one library where it would matter. The deployment image is reproducible
# through its base image digest and this file together.

uvicorn[standard]>=0.27

PyMuPDF>=1.24                # PDF parsing and native vector path extraction

opencv-python-headless>=4.9  # scan conversion and raster tracing only; never the GUI build
pillow>=10.0
ezdxf>=1.4                   # DXF reading, and DWG once converted

# Persistence. Optional at runtime: with no DATABASE_URL the application runs and
# simply cannot save. psycopg 3 rather than psycopg2 for its built-in pool and its
# native support for setting a connection's search_path at connect time, which is
# how this application guarantees it only ever writes to its own schema.
psycopg[binary,pool]>=3.1
python-dotenv>=1.0           # reads a local .env in development; no-op in production
